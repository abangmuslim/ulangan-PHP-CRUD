
CREATE DATABASE IF NOT EXISTS rumahsakitahmadi6;
USE rumahsakitahmadi6;

-- =====================================================
-- 1. HAPUS SELURUH TABEL LAMA
-- =====================================================

SET FOREIGN_KEY_CHECKS = 0;

DROP TABLE IF EXISTS pendaftarancheckup;
DROP TABLE IF EXISTS detilkunjungan;
DROP TABLE IF EXISTS kunjungan;
DROP TABLE IF EXISTS obat;
DROP TABLE IF EXISTS merk;
DROP TABLE IF EXISTS kategori;
DROP TABLE IF EXISTS jadwaldokter;
DROP TABLE IF EXISTS dokter;
DROP TABLE IF EXISTS pasien;
DROP TABLE IF EXISTS `user`;

SET FOREIGN_KEY_CHECKS = 1;


-- =====================================================
-- 2. TABEL USER
-- =====================================================

CREATE TABLE `user` (
    iduser INT AUTO_INCREMENT PRIMARY KEY,
    namauser VARCHAR(100) NOT NULL,
    username VARCHAR(50) NOT NULL UNIQUE,
    password VARCHAR(255) NOT NULL,
    nohp VARCHAR(15),
    alamat TEXT,
    foto VARCHAR(255)
) ENGINE=InnoDB;


-- =====================================================
-- 3. TABEL PASIEN
-- =====================================================

CREATE TABLE pasien (
    idpasien INT AUTO_INCREMENT PRIMARY KEY,
    namapasien VARCHAR(100) NOT NULL,
    username VARCHAR(50) UNIQUE,
    password VARCHAR(255),
    nohp VARCHAR(15),
    alamat TEXT,
    foto VARCHAR(255)
) ENGINE=InnoDB;


-- =====================================================
-- 4. TABEL DOKTER
-- =====================================================

CREATE TABLE dokter (
    iddokter INT AUTO_INCREMENT PRIMARY KEY,
    namadokter VARCHAR(100) NOT NULL,
    spesialis VARCHAR(100),
    nohp VARCHAR(15),
    alamat TEXT,
    foto VARCHAR(255)
) ENGINE=InnoDB;


-- =====================================================
-- 5. TABEL JADWAL DOKTER
-- =====================================================

CREATE TABLE jadwaldokter (
    idjadwal INT AUTO_INCREMENT PRIMARY KEY,
    iddokter INT NOT NULL,
    haripraktik VARCHAR(15) NOT NULL,
    jam_mulai TIME NOT NULL,
    jam_selesai TIME NOT NULL,

    FOREIGN KEY (iddokter)
        REFERENCES dokter(iddokter)
        ON UPDATE CASCADE
        ON DELETE RESTRICT
) ENGINE=InnoDB;


-- =====================================================
-- 6. TABEL KATEGORI
-- =====================================================

CREATE TABLE kategori (
    idkategori INT AUTO_INCREMENT PRIMARY KEY,
    namakategori VARCHAR(100) NOT NULL
) ENGINE=InnoDB;


-- =====================================================
-- 7. TABEL MERK
-- =====================================================

CREATE TABLE merk (
    idmerk INT AUTO_INCREMENT PRIMARY KEY,
    namamerk VARCHAR(100) NOT NULL
) ENGINE=InnoDB;


-- =====================================================
-- 8. TABEL OBAT
-- =====================================================

CREATE TABLE obat (
    idobat INT AUTO_INCREMENT PRIMARY KEY,
    idmerk INT NOT NULL,
    idkategori INT NOT NULL,
    namaobat VARCHAR(100) NOT NULL,
    harga INT DEFAULT 0,
    tanggalstok DATE,
    stok INT DEFAULT 0,
    foto VARCHAR(255),

    FOREIGN KEY (idmerk)
        REFERENCES merk(idmerk)
        ON UPDATE CASCADE
        ON DELETE RESTRICT,

    FOREIGN KEY (idkategori)
        REFERENCES kategori(idkategori)
        ON UPDATE CASCADE
        ON DELETE RESTRICT
) ENGINE=InnoDB;


-- =====================================================
-- 9. TABEL KUNJUNGAN
-- =====================================================

CREATE TABLE kunjungan (
    idkunjungan INT AUTO_INCREMENT PRIMARY KEY,
    iduser INT NOT NULL,
    idpasien INT NOT NULL,
    iddokter INT NOT NULL,
    idjadwal INT NOT NULL,

    tanggalkunjungan DATE NOT NULL,
    keluhan TEXT,
    diagnosis TEXT,

    totalkunjungan INT DEFAULT 0,

    jumlahbayar INT NOT NULL DEFAULT 0,
    kembalian INT NOT NULL DEFAULT 0,

    statuspembayaran ENUM(
        'Belum Bayar',
        'Lunas'
    ) NOT NULL DEFAULT 'Belum Bayar',

    statuskunjungan ENUM(
        'Proses',
        'Selesai'
    ) NOT NULL DEFAULT 'Proses',

    FOREIGN KEY (iduser)
        REFERENCES `user`(iduser)
        ON UPDATE CASCADE
        ON DELETE RESTRICT,

    FOREIGN KEY (idpasien)
        REFERENCES pasien(idpasien)
        ON UPDATE CASCADE
        ON DELETE RESTRICT,

    FOREIGN KEY (iddokter)
        REFERENCES dokter(iddokter)
        ON UPDATE CASCADE
        ON DELETE RESTRICT,

    FOREIGN KEY (idjadwal)
        REFERENCES jadwaldokter(idjadwal)
        ON UPDATE CASCADE
        ON DELETE RESTRICT
) ENGINE=InnoDB;


-- =====================================================
-- 10. TABEL DETIL KUNJUNGAN
-- =====================================================

CREATE TABLE detilkunjungan (
    iddetilkunjungan INT AUTO_INCREMENT PRIMARY KEY,
    idkunjungan INT NOT NULL,
    idobat INT NOT NULL,
    jumlah INT NOT NULL DEFAULT 1,
    harga INT NOT NULL DEFAULT 0,
    subtotal INT NOT NULL DEFAULT 0,
    aturanpakai VARCHAR(255),

    FOREIGN KEY (idkunjungan)
        REFERENCES kunjungan(idkunjungan)
        ON UPDATE CASCADE
        ON DELETE RESTRICT,

    FOREIGN KEY (idobat)
        REFERENCES obat(idobat)
        ON UPDATE CASCADE
        ON DELETE RESTRICT
) ENGINE=InnoDB;


-- =====================================================
-- 11. TABEL PENDAFTARAN CHECKUP
-- =====================================================

CREATE TABLE pendaftarancheckup (
    idpendaftaran INT AUTO_INCREMENT PRIMARY KEY,
    idpasien INT NOT NULL,
    idjadwal INT NOT NULL,

    idkunjungan INT UNIQUE DEFAULT NULL,

    tanggalperiksa DATE NOT NULL,
    keluhan TEXT NOT NULL,

    status ENUM(
        'Menunggu',
        'Disetujui',
        'Ditolak'
    ) NOT NULL DEFAULT 'Menunggu',

    catatanadmin TEXT,

    tanggalpengajuan DATETIME
        NOT NULL DEFAULT CURRENT_TIMESTAMP,

    FOREIGN KEY (idpasien)
        REFERENCES pasien(idpasien)
        ON UPDATE CASCADE
        ON DELETE RESTRICT,

    FOREIGN KEY (idjadwal)
        REFERENCES jadwaldokter(idjadwal)
        ON UPDATE CASCADE
        ON DELETE RESTRICT,

    FOREIGN KEY (idkunjungan)
        REFERENCES kunjungan(idkunjungan)
        ON UPDATE CASCADE
        ON DELETE RESTRICT
) ENGINE=InnoDB;
